name: Unreal Build (ALSAEQA Mobile)

# Runs entirely on GitHub-hosted Linux runners using Epic Games' official
# Unreal Engine container image (ghcr.io/epicgames/unreal-engine). This
# replaces the previous approach (bare Windows runner with no engine
# installed), which could never succeed because Unreal Engine itself was
# never present on the machine running the job.
#
# Requires: the GitHub account running this repo must be a member of the
# EpicGames GitHub org (linked via unrealengine.com/account -> Linked
# accounts -> GitHub) so that GITHUB_TOKEN can pull the private image from
# ghcr.io/epicgames/unreal-engine.
#
# KNOWN OPEN RISK (documented honestly, not hidden):
#   Android packaging inside this container is not guaranteed to work out
#   of the box — the base dev image does not ship the Android SDK/NDK.
#   This workflow installs them manually. If UE5.8's expected NDK version
#   differs from the one pinned below, the Android step may need a version
#   bump. Check the failing step's log for the exact NDK version Unreal
#   requests and adjust ANDROID_NDK_VERSION accordingly.

on:
  workflow_dispatch:
    inputs:
      skip_android:
        description: "Skip Android packaging step (import/verify only)"
        type: boolean
        default: false
  push:
    branches: [main]
    paths:
      - 'Content/**'
      - 'Source/**'
      - 'Config/**'
      - 'Tools/**'
      - '.github/workflows/unreal-build.yml'

env:
  UE_IMAGE: ghcr.io/epicgames/unreal-engine:dev-5.8
  PROJECT_FILE: ALSAEQA.uproject
  ANDROID_NDK_VERSION: "27.2.12479018"
  ANDROID_API_LEVEL: "34"

jobs:
  import-and-build:
    runs-on: ubuntu-latest
    timeout-minutes: 300
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Log in to GitHub Container Registry
        run: echo "${{ secrets.GITHUB_TOKEN }}" | docker login ghcr.io -u "${{ github.actor }}" --password-stdin

      - name: Pull Unreal Engine container image
        run: docker pull "$UE_IMAGE"

      # ---------------------------------------------------------------
      # Stage 1: Download + verify the real-art release assets
      # ---------------------------------------------------------------
      - name: Download and verify release assets
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p Downloads
          python3 - <<'PY'
import json, hashlib, urllib.request, os, sys

with open("Tools/Assets/assets-manifest.json") as f:
    manifest = json.load(f)

tag = manifest["release_tag"]
repo = os.environ.get("GITHUB_REPOSITORY", "ali12hhh-oss/alsaeqa_mobile")
# Assets were originally published under ali12hhh-oss/alsaeqa's release;
# fall back there if not found in this repo's own releases.
candidate_repos = [repo, "ali12hhh-oss/alsaeqa"]

for asset in manifest["assets"]:
    fname = asset["file"]
    expected = asset["sha256"]
    dest = os.path.join("Downloads", fname)
    downloaded = False
    for r in candidate_repos:
        url = f"https://github.com/{r}/releases/download/{tag}/{fname}"
        try:
            print(f"Trying {url}")
            urllib.request.urlretrieve(url, dest)
            downloaded = True
            break
        except Exception as e:
            print(f"  failed: {e}")
    if not downloaded:
        sys.exit(f"Could not download {fname} from any candidate repo")

    h = hashlib.sha256()
    with open(dest, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    digest = h.hexdigest()
    if digest != expected:
        sys.exit(f"Checksum mismatch for {fname}: expected {expected}, got {digest}")
    print(f"OK  {fname}  sha256={digest}")
PY

      - name: Expand nested source archives
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p Content/Art/_staging
          for z in Downloads/*.zip; do
            echo "Expanding $z"
            unzip -q -o "$z" -d Content/Art/_staging
          done
          find Content/Art/_staging -maxdepth 2 -name '*.zip' -print0 | \
            while IFS= read -r -d '' nested; do
              unzip -q -o "$nested" -d "$(dirname "$nested")"
            done

      # ---------------------------------------------------------------
      # Stage 2: Run Unreal Editor commandlets inside the container
      # ---------------------------------------------------------------
      - name: Import assets into Unreal (ImportReleaseArt)
        shell: bash
        run: |
          set -euo pipefail
          docker run --rm \
            -v "$PWD:/project" \
            -w /project \
            "$UE_IMAGE" \
            /home/ue4/UnrealEngine/Engine/Binaries/Linux/UnrealEditor-Cmd \
              "/project/$PROJECT_FILE" \
              -run=pythonscript \
              -script="/project/Tools/Unreal/ImportReleaseArt.py" \
              -unattended -nopause -nosplash -stdout -FullStdOutLogOutput

      - name: Wire animations, camera, and world (best-effort automation)
        shell: bash
        run: |
          set -euo pipefail
          docker run --rm \
            -v "$PWD:/project" \
            -w /project \
            "$UE_IMAGE" \
            /home/ue4/UnrealEngine/Engine/Binaries/Linux/UnrealEditor-Cmd \
              "/project/$PROJECT_FILE" \
              -run=pythonscript \
              -script="/project/Tools/Unreal/WireGameplayAndCamera.py" \
              -unattended -nopause -nosplash -stdout -FullStdOutLogOutput

      - name: Commit imported assets back to the repository
        shell: bash
        run: |
          set -euo pipefail
          git config user.name "alsaeqa-ci"
          git config user.email "ci@users.noreply.github.com"
          git add Content/
          if git diff --cached --quiet; then
            echo "No content changes to commit"
          else
            git commit -m "CI: import real art assets and wire gameplay [skip ci]"
            git push
          fi

      # ---------------------------------------------------------------
      # Stage 3: Android packaging (experimental — see header notes)
      # ---------------------------------------------------------------
      - name: Set up Android SDK/NDK
        if: ${{ inputs.skip_android != true }}
        shell: bash
        run: |
          set -euo pipefail
          docker run --rm \
            -v "$PWD:/project" \
            -w /project \
            "$UE_IMAGE" \
            bash -c "
              set -e
              yes | sdkmanager --licenses > /dev/null || true
              sdkmanager \"platform-tools\" \"platforms;android-$ANDROID_API_LEVEL\" \"ndk;$ANDROID_NDK_VERSION\" \"build-tools;34.0.0\"
            "

      - name: Package Android build
        if: ${{ inputs.skip_android != true }}
        shell: bash
        run: |
          set -euo pipefail
          docker run --rm \
            -v "$PWD:/project" \
            -w /project \
            "$UE_IMAGE" \
            /home/ue4/UnrealEngine/Engine/Build/BatchFiles/RunUAT.sh BuildCookRun \
              -project="/project/$PROJECT_FILE" \
              -platform=Android \
              -clientconfig=Development \
              -cook -build -stage -package \
              -archive -archivedirectory="/project/Builds"

      - name: Upload Android build artifact
        if: ${{ inputs.skip_android != true }}
        uses: actions/upload-artifact@v4
        with:
          name: alsaeqa-android-build
          path: Builds/**
          retention-days: 14
